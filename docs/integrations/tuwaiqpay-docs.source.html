المصادقة — Authentication

الخطوة الأولى في أي تكامل. احصل على access_token باستخدام بيانات الاعتماد.

POST /api/v1/auth/authenticate

Headers

HTTP
Content-Type: application/json Accept: application/json X-Language: ar

Request Body

username string مطلوب
رقم الجوال (بصيغة +966XXXXXXXXX) أو البريد الإلكتروني
userNameType enum مطلوب
MOBILE أو EMAIL
password string مطلوب
كلمة المرور الخاصة بحسابك
📱

تنسيق رقم الجوال: يجب أن يبدأ بـ + متبوعاً برمز الدولة. مثال: +966543751523. الصيغ المحلية مثل 05XXXXXXXX غير مقبولة مباشرة.

أمثلة الكود

PHP
function authenticateTuwaiqPay($username, $password, $type = 'MOBILE') { $baseUrl = 'https://onboarding-uat.tuwaiqpay.com.sa/api/v1'; $data = [ 'username' => $username, 'userNameType' => $type, 'password' => $password, ]; $ch = curl_init($baseUrl . '/auth/authenticate'); curl_setopt($ch, CURLOPT_RETURNTRANSFER, true); curl_setopt($ch, CURLOPT_POST, true); curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data)); curl_setopt($ch, CURLOPT_HTTPHEADER, [ 'Content-Type: application/json', 'Accept: application/json', 'X-Language: ar', ]); $response = curl_exec($ch); $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE); curl_close($ch); $result = json_decode($response, true); if ($httpCode === 200) { return $result['data']['access_token']; } throw new Exception($result['message'] ?? 'Auth failed'); }
Python
import requests def authenticate(username, password, username_type="MOBILE"): base_url = "https://onboarding-uat.tuwaiqpay.com.sa/api/v1" resp = requests.post( f"{base_url}/auth/authenticate", json={ "username": username, "userNameType": username_type, "password": password, }, headers={ "Content-Type": "application/json", "Accept": "application/json", "X-Language": "ar", }, ) resp.raise_for_status() return resp.json()["data"]["access_token"]
Node.js
const axios = require('axios'); async function authenticate(username, password, type = 'MOBILE') { const baseUrl = 'https://onboarding-uat.tuwaiqpay.com.sa/api/v1'; const { data } = await axios.post( `${baseUrl}/auth/authenticate`, { username, userNameType: type, password }, { headers: { 'Content-Type': 'application/json', 'Accept': 'application/json', 'X-Language': 'ar', }, } ); return data.data.access_token; }

إنشاء فاتورة — Create Bill

بعد المصادقة، أنشئ فاتورة دفع وأرسل رابطها للعميل.

POST /api/v1/integration/bills

Headers

HTTP
Authorization: Bearer {access_token} Content-Type: application/json Accept: application/json X-Language: ar

Parameters

amountfloatمطلوب
المبلغ بالريال السعودي. مثال: 150.00
actionDateInDaysintegerمطلوب
عدد أيام صلاحية الفاتورة (1–365)
currencyIdintegerمطلوب
1 = SAR  |  2 = USD  |  3 = EUR
supportedPaymentMethodsstring[]مطلوب
وسائل الدفع: VISA MASTER MADA AMEX STC_PAY APPLE_PAY
descriptionstringمطلوب
وصف الفاتورة. مثال: "طلب #12345"
customerNamestringمطلوب
اسم العميل
customerMobilePhonestringمطلوب
جوال العميل بصيغة +966XXXXXXXXX
customerEmailstringاختياري
بريد العميل الإلكتروني
includeVatbooleanاختياري
هل يشمل ضريبة القيمة المضافة؟ الافتراضي: false

أمثلة الكود

PHP
function createBill($token, $amount, $desc, $name, $phone) { $baseUrl = 'https://onboarding-uat.tuwaiqpay.com.sa/api/v1'; $payload = [ 'actionDateInDays' => 3, 'amount' => (float) $amount, 'currencyId' => 1, 'supportedPaymentMethods' => ['VISA','MASTER','MADA','STC_PAY','APPLE_PAY'], 'description' => $desc, 'customerName' => $name, 'customerMobilePhone' => $phone, 'includeVat' => false, ]; $ch = curl_init($baseUrl . '/integration/bills'); curl_setopt($ch, CURLOPT_RETURNTRANSFER, true); curl_setopt($ch, CURLOPT_POST, true); curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($payload)); curl_setopt($ch, CURLOPT_HTTPHEADER, [ 'Authorization: Bearer ' . $token, 'Content-Type: application/json', 'Accept: application/json', 'X-Language: ar', ]); $response = json_decode(curl_exec($ch), true); curl_close($ch); return $response['data']; // data.link → رابط الدفع للعميل // data.merchantTransactionId → احفظه في DB }
Python
def create_bill(token, amount, desc, name, phone): resp = requests.post( f"{BASE_URL}/integration/bills", json={ "actionDateInDays": 3, "amount": float(amount), "currencyId": 1, "supportedPaymentMethods": [ "VISA", "MASTER", "MADA", "STC_PAY", "APPLE_PAY" ], "description": desc, "customerName": name, "customerMobilePhone": phone, "includeVat": False, }, headers={ "Authorization": f"Bearer {token}", "Content-Type": "application/json", "X-Language": "ar", }, ) resp.raise_for_status() return resp.json()["data"]
Node.js
async function createBill(token, amount, desc, name, phone) { const { data } = await axios.post( `${BASE_URL}/integration/bills`, { actionDateInDays: 3, amount: parseFloat(amount), currencyId: 1, supportedPaymentMethods: [ 'VISA', 'MASTER', 'MADA', 'STC_PAY', 'APPLE_PAY' ], description: desc, customerName: name, customerMobilePhone: phone, includeVat: false, }, { headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json', 'X-Language': 'ar', }, } ); return data.data; // data.link → Payment URL for customer // data.merchantTransactionId → Save to DB }

الإعداد

📋

ملاحظة مهمة: لا توجد لوحة تحكم (Dashboard) للتجار حالياً — يتم إدارة الحساب عبر التطبيق فقط. لإضافة Webhook URL، يجب التواصل مع الدعم الفني وإرسال الرابط المطلوب ليتم تفعيله من طرفهم.

خطوات الإعداد
// 1. جهّز Webhook URL على سيرفرك // مثال: https://yourdomain.com/tuwaiqpay-webhook // 2. تواصل مع الدعم الفني: // 📧 support@tuwaiqpay.com.sa // 💬 wa.me/966509684000 // 3. أرسل لهم: // • Webhook URL الخاص بك // • الأحداث المطلوبة (Payment Completed, Failed, Cancelled) // • Merchant ID الخاص بك // 4. انتظر التأكيد من الدعم الفني بتفعيل الـ Webhook

PHP
// webhook.php $payload = file_get_contents('php://input'); $data = json_decode($payload, true); $tx = $data['transactionDetails'] ?? []; $merchantTxId = $tx['merchantTransactionId'] ?? ''; $status = $tx['transactionStatus'] ?? ''; $amount = $tx['bill']['amount'] ?? 0; // Map TuwaiqPay status → your status $map = [ 'SETTLED' => 'completed', 'APPROVED' => 'completed', 'PENDING_SETTLEMENT' => 'completed', 'FAILED' => 'failed', 'DECLINED' => 'failed', 'CANCELLED' => 'cancelled', ]; $newStatus = $map[$status] ?? 'pending'; // Update your database $stmt = $pdo->prepare(" UPDATE payments SET status = ?, tuwaiqpay_status = ?, updated_at = NOW() WHERE merchant_transaction_id = ? "); $stmt->execute([$newStatus, $status, $merchantTxId]); http_response_code(200); echo json_encode(['success' => true]);

Token Caching — PHP
// Cache the access token — don't re-authenticate on every request $_SESSION['tp_token'] = $token; $_SESSION['tp_expires'] = time() + 3500; // 3600 - 100s buffer if (isset($_SESSION['tp_token']) && time() < $_SESSION['tp_expires']) { $token = $_SESSION['tp_token']; } else { $token = authenticateTuwaiqPay(...); }

الأسئلة الشائعة

كيف أحصل على حساب TuwaiqPay؟▼
زر tuwaiqpay.com.sa وسجّل كتاجر. ستحتاج بريدك ورقم جوالك ونوع النشاط.
هل يوجد رسوم للتكامل؟▼
التكامل مجاني، لكن توجد عمولة على المعاملات حسب اتفاقيتك مع TuwaiqPay.
لماذا أحصل على HTTP 403؟▼
حسابك لا يملك صلاحيات API بعد. تواصل مع support@tuwaiqpay.com.sa واطلب تفعيل API Integration.
كيف أتعامل مع expireDate في UAT؟▼
أضف +3 ساعات عند حفظه في قاعدة البيانات. هذا بسبب مشكلة timezone في بيئة UAT.
هل Webhook ضروري؟▼
نعم! بدونه لن تتحدث حالة الطلبات تلقائياً عند الدفع أو الفشل. لإضافة Webhook URL، تواصل مع الدعم الفني وأرسل لهم الرابط ليتم تفعيله (لا توجد لوحة تحكم حالياً — فقط التطبيق).
كيف أختبر Webhook محلياً؟▼
استخدم ngrok أو localtunnel لإنشاء URL عام مؤقت يوجه الطلبات لجهازك المحلي.
ما هي وسائل الدفع المدعومة؟▼
MADA, VISA, Mastercard, Apple Pay, STC Pay, و American Express.
هل يوجد حد أقصى للمبلغ؟▼
يعتمد على إعدادات حسابك. تواصل مع TuwaiqPay لمعرفة الحدود.
